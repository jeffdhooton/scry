func semTime(day int) time.Time { return time.Date(2026, 8, day, 0, 0, 0, 0, time.UTC) }
func semEntities(t *testing.T, st *store.Store) {
	t.Helper()
	for _, name := range []string{"Atlas", "Borealis", "Cygnus", "Draco", "Equuleus"} {
		if err := st.PutEntity(store.Entity{Slug: store.Slugify(name), Name: name, Type: "service", CreatedAt: semTime(1)}); err != nil {
			t.Fatal(err)
		}
	}
}
func semPut(t *testing.T, st *store.Store, f store.Fact) {
	t.Helper()
	if err := st.PutFact(f); err != nil {
		t.Fatal(err)
	}
}
func semApply(t *testing.T, st *store.Store, at time.Time, fs ...extract.Fct) (Stats, error) {
	t.Helper()
	return Apply(st, store.Episode{ID: "incoming", OccurredAt: at}, "", extract.Result{Facts: fs}, DefaultExclusive)
}
func semCurrent(t *testing.T, st *store.Store, src string) []store.Fact {
	t.Helper()
	fs, err := st.FactsFrom(src, false)
	if err != nil {
		t.Fatal(err)
	}
	return fs
}
func semFact(dst, relation string, at time.Time) extract.Fct {
	return extract.Fct{Src: "Atlas", Relation: relation, Dst: dst, Fact: "Atlas selects " + dst + ".", ValidFrom: at.Format(time.RFC3339Nano), Confidence: .8}
}

// Equal-start contradictory inputs cannot establish a unique exclusive state.
// Both orders are required; no winner can be justified by slice position.
func TestSemanticExclusiveEqualStartMustNotCommitTwoCurrentTargets(t *testing.T) {
	for _, reverse := range []bool{false, true} {
		t.Run(fmt.Sprint(reverse), func(t *testing.T) {
			st := openTemp(t)
			semEntities(t, st)
			old := store.Fact{Src: "atlas", Relation: "replaced_by", Dst: "borealis", Fact: "An earlier Borealis assertion.", ValidFrom: semTime(1), Confidence: .9, Episodes: []string{"old"}}
			semPut(t, st, old)
			a, b := semFact("Borealis", "replaced_by", semTime(10)), semFact("Cygnus", "replaced_by", semTime(10))
			// Both undated inputs use the episode instant, as ordinary extraction does.
			a.ValidFrom, b.ValidFrom = "", ""
			fs := []extract.Fct{a, b}
			if reverse {
				fs[0], fs[1] = fs[1], fs[0]
			}
			stats, err := semApply(t, st, semTime(10), fs...)
			current := semCurrent(t, st, "atlas")
			t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
			if err == nil && len(current) > 1 {
				t.Fatalf("contradictory exclusive targets committed current: %+v", current)
			}
		})
	}
}

func TestSemanticExclusiveExplicitStartsMustNotChooseSliceWinner(t *testing.T) {
	for _, reverse := range []bool{false, true} {
		t.Run(fmt.Sprint(reverse), func(t *testing.T) {
			st := openTemp(t)
			semEntities(t, st)
			fs := []extract.Fct{semFact("Borealis", "replaced_by", semTime(5)), semFact("Cygnus", "replaced_by", semTime(10))}
			if reverse {
				fs[0], fs[1] = fs[1], fs[0]
			}
			stats, err := semApply(t, st, semTime(20), fs...)
			t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
			if err != nil {
				return
			} // Conservative ambiguity refusal is preservation-safe.
			cur := semCurrent(t, st, "atlas")
			if len(cur) != 1 || cur[0].Dst != "cygnus" {
				t.Fatalf("later explicit state lost to slice order: %+v", cur)
			}
			for _, f := range mustFacts(t, st, "atlas") {
				if f.Dst == "borealis" && (f.InvalidAt == nil || !f.InvalidAt.Equal(semTime(10))) {
					t.Errorf("episode time invented end instead of explicit successor start: %+v", f)
				}
			}
		})
	}
}

func TestSemanticExclusiveLaterObservationOfOlderStateMustNotRetireNewer(t *testing.T) {
	st := openTemp(t)
	semEntities(t, st)
	old := store.Fact{Src: "atlas", Relation: "replaced_by", Dst: "cygnus", Fact: "Cygnus began on August 10.", ValidFrom: semTime(10), Episodes: []string{"newer"}, Confidence: .9}
	semPut(t, st, old)
	stats, err := semApply(t, st, semTime(20), semFact("Borealis", "replaced_by", semTime(5)))
	t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
	if err == nil {
		cur := semCurrent(t, st, "atlas")
		if len(cur) != 1 || cur[0].Dst != "cygnus" {
			t.Fatalf("older explicit assertion replaced newer state: %+v", cur)
		}
	}
}

func TestSemanticExclusiveFullLiteralTargetsMustDiffer(t *testing.T) {
	st := openTemp(t)
	semEntities(t, st)
	if store.AttrDst("stage_ready") != store.AttrDst("stage-ready") {
		t.Fatal("fixture must collide only after normalization")
	}
	old := store.Fact{Src: "atlas", Relation: "status", Value: "stage_ready", Fact: "The literal state is stage_ready.", ValidFrom: semTime(1), Episodes: []string{"old"}, Confidence: .9}
	semPut(t, st, old)
	stats, err := semApply(t, st, semTime(10), semFact("stage-ready", "status", semTime(10)))
	t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
	if err == nil && len(semCurrent(t, st, "atlas")) != 1 {
		t.Fatal("distinct full literals were treated as one exclusive target")
	}
}

func TestSemanticSupersedesAmbiguousEligibleCanonicalMustNotChooseFirst(t *testing.T) {
	st := openTemp(t)
	semEntities(t, st)
	for _, day := range []int{1, 5} {
		semPut(t, st, store.Fact{Src: "atlas", Relation: "uses", Dst: "borealis", Fact: fmt.Sprintf("Distinct purpose %d.", day), ValidFrom: semTime(day), Episodes: []string{"old"}, Confidence: .9})
	}
	before := mustFacts(t, st, "atlas")
	f := semFact("Cygnus", "uses", semTime(10))
	f.Supersedes = &extract.SupRef{Src: "Atlas", Relation: "uses", Dst: "Borealis"}
	stats, err := semApply(t, st, semTime(10), f)
	t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
	if !errors.Is(err, store.ErrFactConflict) || !reflect.DeepEqual(before, mustFacts(t, st, "atlas")) {
		t.Fatal("ambiguous triple selected one assertion and committed its dependent input")
	}
}

func TestSemanticSupersedesEligibilityBeforeUniqueness(t *testing.T) {
	for _, relation := range []string{"uses", "synthesizes_with"} {
		t.Run(relation, func(t *testing.T) {
			st := openTemp(t)
			semEntities(t, st)
			past := time.Date(1971, 1, 1, 0, 0, 0, 0, time.UTC)
			future := time.Date(2002, 1, 1, 0, 0, 0, 0, time.UTC)
			episode := time.Date(1980, 1, 1, 0, 0, 0, 0, time.UTC)
			canonical, _ := Map(relation)
			raw := ""
			if canonical != relation {
				raw = relation
			}
			for _, at := range []time.Time{past, future} {
				semPut(t, st, store.Fact{Src: "atlas", Relation: canonical, Dst: "borealis", RawRelation: raw, Fact: "Version at " + at.Format(time.RFC3339), ValidFrom: at, Episodes: []string{"old"}, Confidence: .9})
			}
			fs := mustFacts(t, st, "atlas")
			if !fs[0].ValidFrom.Equal(future) {
				t.Fatal("fixture requires first raw key to be ineligible")
			}
			f := semFact("Cygnus", "uses", episode)
			f.Supersedes = &extract.SupRef{Src: "Atlas", Relation: relation, Dst: "Borealis"}
			stats, err := semApply(t, st, episode, f)
			t.Logf("stats=%+v err=%v all=%+v", stats, err, mustFacts(t, st, "atlas"))
			if err != nil {
				t.Fatalf("one eligible exact candidate blocked by ineligible candidate: %v", err)
			}
			for _, fact := range mustFacts(t, st, "atlas") {
				if fact.Dst == "borealis" && fact.ValidFrom.Equal(past) && (fact.InvalidAt == nil || !fact.InvalidAt.Equal(episode)) {
					t.Error("unique eligible candidate missed because first candidate was ineligible")
				}
				if fact.Dst == "borealis" && fact.ValidFrom.Equal(future) && fact.InvalidAt != nil {
					t.Error("ineligible future fact invalidated")
				}
			}
		})
	}
}

func TestSemanticMalformedFreshDateWithholdsUnrelatedFact(t *testing.T) {
	st := openTemp(t)
	semEntities(t, st)
	f := semFact("Borealis", "uses", semTime(10))
	f.ValidFrom = "malformed"
	unrelated := extract.Fct{Src: "Draco", Relation: "uses", Dst: "Equuleus", Fact: "Draco uses Equuleus independently.", ValidFrom: semTime(10).Format(time.RFC3339), Confidence: .8}
	stats, err := semApply(t, st, semTime(10), f, unrelated)
	if !errors.Is(err, store.ErrFactConflict) || stats != (Stats{}) || len(mustFacts(t, st, "draco")) != 0 {
		t.Fatalf("characterization changed: %+v %v", stats, err)
	}
	t.Log("Fresh malformed date with vacant address blocks 1/1 unrelated valid assertion; no existing identity could have been falsely matched.")
}

func TestSemanticExactRepeatPreservesCurrentAndHistorical(t *testing.T) {
	for _, historical := range []bool{false, true} {
		t.Run(fmt.Sprint(historical), func(t *testing.T) {
			st := openTemp(t)
			semEntities(t, st)
			old := store.Fact{Src: "atlas", Relation: "uses", Dst: "borealis", Fact: "An exact assertion.", ValidFrom: semTime(1), Confidence: .95, Episodes: []string{"origin"}}
			if historical {
				end := semTime(3)
				old.InvalidAt = &end
			}
			semPut(t, st, old)
			f := semFact("Borealis", "uses", semTime(1))
			f.Fact = old.Fact
			f.Confidence = .2
			f.ValidFrom = "2026-08-01T02:00:00+02:00"
			stats, err := semApply(t, st, semTime(10), f, f)
			if err != nil {
				t.Fatal(err)
			}
			old.Episodes = append(old.Episodes, "incoming")
			if stats.FactsMerged != 2 || stats.FactsAdded != 0 || !reflect.DeepEqual(mustFacts(t, st, "atlas"), []store.Fact{old}) {
				t.Fatalf("metadata/start/history changed: stats=%+v facts=%+v", stats, mustFacts(t, st, "atlas"))
			}
		})
	}
}

// All raw access below is restricted to this test's t.TempDir fixture.
func semRawFixture(t *testing.T, relation string) (*store.Store, string, []byte, []byte, store.Fact) {
	t.Helper()
	dir := filepath.Join(t.TempDir(), "raw-badger")
	st, err := store.Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	semEntities(t, st)
	f := store.Fact{Src: "atlas", Relation: relation, Dst: "borealis", Fact: "An original assertion with future metadata.", ValidFrom: semTime(1), Confidence: .9, Episodes: []string{"origin"}}
	semPut(t, st, f)
	if err := st.Close(); err != nil {
		t.Fatal(err)
	}
	k := []byte(fmt.Sprintf("fa:%s:%s:%s:%d", f.Src, f.Relation, f.KeyDst(), f.ValidFrom.UnixNano()))
	raw, err := json.Marshal(f)
	if err != nil {
		t.Fatal(err)
	}
	raw = append(raw[:len(raw)-1], []byte(",\"future_metadata\":{\"keep\":\"exact\"}}")...)
	db, err := badger.Open(badger.DefaultOptions(dir).WithLogger(nil).WithCompression(0))
	if err != nil {
		t.Fatal(err)
	}
	if err := db.Update(func(txn *badger.Txn) error { return txn.Set(k, raw) }); err != nil {
		t.Fatal(err)
	}
	if err := db.Close(); err != nil {
		t.Fatal(err)
	}
	st, err = store.Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	return st, dir, k, raw, f
}
func semReadClosedRaw(t *testing.T, st *store.Store, dir string, k []byte) []byte {
	t.Helper()
	if err := st.Close(); err != nil {
		t.Fatal(err)
	}
	db, err := badger.Open(badger.DefaultOptions(dir).WithLogger(nil).WithCompression(0))
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	var raw []byte
	if err := db.View(func(txn *badger.Txn) error {
		item, e := txn.Get(k)
		if e != nil {
			return e
		}
		raw, e = item.ValueCopy(nil)
		return e
	}); err != nil {
		t.Fatal(err)
	}
	return raw
}
func TestSemanticUnknownRawPreservedOnEveryResolverMutation(t *testing.T) {
	for _, route := range []string{"exact-restatement", "exclusive", "supersedes"} {
		t.Run(route, func(t *testing.T) {
			relation := "uses"
			if route == "exclusive" {
				relation = "replaced_by"
			}
			st, dir, key, before, old := semRawFixture(t, relation)
			f := semFact("Cygnus", relation, semTime(10))
			if route == "exact-restatement" {
				f = semFact("Borealis", relation, old.ValidFrom)
				f.Fact = old.Fact
			}
			if route == "supersedes" {
				f.Supersedes = &extract.SupRef{Src: "Atlas", Relation: relation, Dst: "Borealis"}
			}
			stats, err := semApply(t, st, semTime(10), f)
			after := semReadClosedRaw(t, st, dir, key)
			t.Logf("route=%s stats=%+v err=%v before=%s after=%s", route, stats, err, before, after)
			if err != nil {
				if !bytes.Equal(before, after) {
					t.Fatal("refusal changed raw bytes")
				}
				return
			}
			if !bytes.Contains(after, []byte("\"future_metadata\":{\"keep\":\"exact\"}")) {
				t.Fatal("resolver invalidation discarded unknown raw metadata")
			}
		})
	}
}
